# Environment `alike` — everything that builds or scores a graph.
#
#   conda create -n alike python=3.10 -y && conda activate alike
#   pip install torch==2.11.0 torchvision==0.26.0 \
#       --index-url https://download.pytorch.org/whl/cu128
#   pip install -r requirements-alike.txt
#
# torch must go in first, from the CUDA index: on plain PyPI it resolves to a
# CPU-only wheel.  The cu128 build covers every current architecture, from Turing
# onwards - any current consumer GPU will do.
#
# Version notes:
#   * numpy 2.x is fine here.  It is NOT fine in `rknn`, which caps at 1.26.4 -
#     that cap is the root of the whole two-environment split (see README).
#   * onnxruntime-gpu is optional.  The accuracy harness runs with
#     providers=["CPUExecutionProvider"] on purpose, so the CPU wheel scores
#     identically; the GPU build is not used by anything here.
#   * onnxscript is what torch's newer export path wants; it arrived as a
#     transitive dependency and is pinned only so the export is reproducible.
